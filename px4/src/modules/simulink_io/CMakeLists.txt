# Separate your globbing to make sure it handles files accurately
file(GLOB SIMULINK_C_SRCS "generated_code/*.c")
file(GLOB SIMULINK_CPP_SRCS "generated_code/*.cpp")

px4_add_module(
    MODULE modules__simulink_io
    MAIN simulink_io
    STACK_MAIN 8192
    SRCS
        simulink_io.cpp             # Your C++ PX4 Module Wrapper
        ${SIMULINK_CPP_SRCS}        # Includes your px4_simulink_glue.cpp
        ${SIMULINK_C_SRCS}          # Your pure C Simulink model logic
    INCLUDES
        generated_code

    MODULE_CONFIG
        module.yaml
        module_stock_missing.yaml
)

# Apply warning suppressions ONLY to the C files generated by Simulink
set_source_files_properties(${SIMULINK_C_SRCS}
    PROPERTIES COMPILE_FLAGS "-Wno-float-equal -Wno-logical-op -Wno-unused-variable -Wno-unused-but-set-variable -Wno-parentheses -Wno-error"
)
